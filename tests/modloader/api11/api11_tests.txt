# A derelict that can fly once it wakes up.
ship "API11 Drifter"
	sprite "ship/star barge"
	attributes
		category "Light Freighter"
		cost 1000
		mass 50
		bunks 2
		drag 2
		hull 2000
		shields 500
		"required crew" 1
		thrust 40
		turn 300
		"energy generation" 10
		"energy capacity" 100
		"fuel capacity" 300
	engine -9 38 1
	engine 9 38 1

fleet "API11 TEST: drifters"
	personality derelict
	government "Pirate"
	variant
		"API11 Drifter"


test "API 11"
	status active
	description "Mod API 11: runtime definitions saved and loaded with the pilot, outfits replaced and moved, and a derelict woken up."
	sequence
		inject "API11 Save"
		call "Load First Savegame"
		# The loader checked the definitions and moved outfits when the pilot loaded.
		assert
			"api11 loads" == 1
			"api11 fail" == 0
			"api11 pass" > 0
			"ship model (all): API11 Hull" == 1
			"outfit: API11 Core" == 1
			"outfit: API11 Battery" == 4
			"outfit: API11 Plate" == 0
		# The mod's planet button: replace outfits.
		input
			key k
		assert
			"api11 replaced" == 1
			"api11 fail" == 0
			"outfit: API11 Core" == 0
			"outfit: API11 Core Plus" == 1
			"outfit: API11 Plate" == 1
			"outfit: API11 Battery" == 3
		# Taking off saves the pilot.
		call "Depart"
		assert
			"flagship model: API11 Hull" == 1
			"flagship attribute: shield generation" == 5000
		label "waiting"
		branch "waiting"
			not "api11 woke"
		assert
			"api11 fail" == 0
		# Load what taking off saved.
		input
			key escape
		call "Load First Savegame"
		assert
			"api11 loads" == 2
			"api11 fail" == 0
			"ship model (all): API11 Hull" == 1
			"outfit: API11 Core Plus" == 1
			"outfit: API11 Plate" == 1
			"outfit: API11 Battery" == 3
		call "Depart"
		assert
			"flagship model: API11 Hull" == 1
			"flagship attribute: shield generation" == 5000
			"api11 fail" == 0
